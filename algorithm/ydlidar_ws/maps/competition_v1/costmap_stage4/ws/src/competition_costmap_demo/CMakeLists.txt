cmake_minimum_required(VERSION 3.8)
project(competition_costmap_demo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
add_executable(costmap_stage4_runner src/costmaps.cpp)
ament_target_dependencies(costmap_stage4_runner rclcpp nav2_costmap_2d)
# 同时兼容 Humble ament_cmake 导出的别名和非命名空间目标。
if(TARGET nav2_costmap_2d::nav2_costmap_2d_core)
  target_link_libraries(costmap_stage4_runner nav2_costmap_2d::nav2_costmap_2d_core)
elseif(TARGET nav2_costmap_2d_core)
  target_link_libraries(costmap_stage4_runner nav2_costmap_2d_core)
endif()
install(TARGETS costmap_stage4_runner DESTINATION lib/${PROJECT_NAME})
ament_package()
